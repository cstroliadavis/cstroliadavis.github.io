export function createSemaphoreQueue(deps: { redis: RedisClient; ttlMs: number }) {
  const _ = {
    activeLocks: new Map<string,  string>(),
    maxConcurrency: 40,
    namespace: 'swarm:deploy:semaphore',
  };

  const acquireSlot = async (instanceId: string): Promise<boolean> => {
    const lockKey = `${_.namespace}:${instanceId}`;
    const acquired = await deps.redis.set(lockKey, 'LOCKED', 'NX', 'PX', deps.ttlMs);
    if (acquired) _.activeLocks.set(instanceId, lockKey);
    return Boolean(acquired);
  };

  return { acquireSlot };
}

-- Multi-Tenant Publishing Pipeline CTE Optimization (Hours -> < 2 Min)
WITH IndexedPayloadSlice AS (
  SELECT p.tenant_id, p.document_id, p.published_at,
         JSONB_PATH_QUERY_FIRST(p.payload, '$.metadata.stationCode') AS station_code
  FROM publishing_events p
  WHERE p.status = 'READY' AND p.published_at >= NOW() - INTERVAL '24 hours'
)
SELECT s.station_code, COUNT(*) AS throughput_count
FROM IndexedPayloadSlice s
GROUP BY s.station_code
ORDER BY throughput_count DESC;

// Reusable Web Worker Higher-Order Function (HOF) Wrapper
export const withWorkerThread = <TArgs extends unknown[], TResult>(
  fn: (...args: TArgs) => TResult,
) => {
  const source = `self.onmessage = (e) => self.postMessage((${fn.toString()})(...e.data));`;
  const blobUrl = URL.createObjectURL(new Blob([source], { type: 'text/javascript' }));

  return (...args: TArgs): Promise<TResult> =>
    new Promise((resolve, reject) => {
      const worker = new Worker(blobUrl);
      worker.onmessage = (event) => {
        worker.terminate();
        resolve(event.data);
      };
      worker.onerror = (err) => {
        worker.terminate();
        reject(err);
      };
      worker.postMessage(args);
    });
};
